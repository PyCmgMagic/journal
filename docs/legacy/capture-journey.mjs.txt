import { createRequire } from "node:module";
import { mkdir, writeFile } from "node:fs/promises";
import { resolve } from "node:path";
import assert from "node:assert/strict";
const require = createRequire(import.meta.url);
const { chromium } = require(process.env.PLAYWRIGHT_MODULE || "playwright");
const output = resolve("output/playwright");
await mkdir(output, { recursive: true });
const browser = await chromium.launch({
  headless: true,
  executablePath: process.env.BROWSER_EXECUTABLE,
  args: ["--enable-unsafe-swiftshader"],
});
const context = await browser.newContext({
  viewport: { width: 1440, height: 900 },
  recordVideo: {
    dir: resolve(output, "motion"),
    size: { width: 1440, height: 900 },
  },
});
const page = await context.newPage();
const errors = [],
  frames = [];
page.on("pageerror", (e) => errors.push(String(e)));
page.on("console", (m) => {
  if (m.type() === "error") errors.push(m.text());
});
const wait = (ms) => new Promise((r) => setTimeout(r, ms));
const shot = (name) =>
  page.screenshot({ path: resolve(output, `story-${name}.png`) });
const camera = async () =>
  JSON.parse(await page.locator("canvas").getAttribute("data-camera"));
try {
  await page.goto(process.env.PREVIEW_URL || "http://127.0.0.1:4321/");
  await page.locator('[data-scene-mode="webgl"]').waitFor();
  await page.mouse.move(720, 450);
  await wait(1000);
  await shot("01-arrival");
  const distance = await page.evaluate(
    () => document.documentElement.scrollHeight - innerHeight,
  );
  let previous = 0;
  for (const progress of [0.08, 0.16, 0.24, 0.31, 0.39, 0.46, 0.52]) {
    const y = Math.round(distance * progress);
    await page.mouse.wheel(0, y - previous);
    previous = y;
    await wait(650);
    frames.push({ progress, camera: await camera() });
    if ([0.24, 0.31, 0.39, 0.52].includes(progress))
      await shot(`approach-${progress}`);
  }
  assert.ok(frames.at(-1).camera[2] < 5.1);
  const stopped = await camera();
  await wait(500);
  assert.ok(Math.abs((await camera())[2] - stopped[2]) < 0.01);
  // Reverse through the same route, then return to the stone without a scene switch.
  await page.mouse.wheel(0, -Math.round(distance * 0.21));
  await wait(900);
  const reverse = await camera();
  assert.ok(Math.abs(reverse[2] - frames[3].camera[2]) < 0.04);
  await page.mouse.wheel(0, Math.round(distance * 0.21));
  await wait(1000);
  await page.locator(".monolith-control").hover();
  await wait(350);
  await page.mouse.down();
  await wait(650);
  await shot("02-charging");
  await page.locator('[data-phase="awakening"]').waitFor();
  await page.mouse.up();
  await wait(1400);
  await shot("wake-return");
  await page.locator('[data-phase="awake"]').waitFor();
  await wait(400);
  await shot("03-awake");
  await page.locator('.world-label[data-category="notes"]').hover();
  await wait(450);
  await shot("05-preview");
  const lamp = await page
      .locator('.world-label[data-category="notes"]')
      .boundingBox(),
    preview = await page.locator(".hover-preview").boundingBox();
  assert.ok(preview.y + preview.height < lamp.y);
  await page.locator('.world-label[data-category="notes"]').click();
  await wait(850);
  await shot("06-category");
  await page.locator(".article-row").first().click();
  await page.locator(".prose h1").waitFor();
  await wait(1000);
  await shot("07-reading");
  await page.getByRole("button", { name: "返回观测站" }).click();
  await page.locator(".content-panel").waitFor({ state: "detached" });
  await wait(700);
  await shot("08-return");
  assert.deepEqual(errors, []);
  await writeFile(
    resolve(output, "story-checks.json"),
    JSON.stringify({ frames, reverse, errors }, null, 2),
  );
} finally {
  const video = page.video();
  await context.close();
  await video.saveAs(resolve(output, "journey-actual.webm"));
  await browser.close();
}
